local function StartCooldown(key, duration)

refs.overlay.Visible = true
refs.fill.Size = UDim2.fromScale(1, 1)
refs.countdown.Visible = true

TweenInfo.new(duration, Enum.EasingStyle.Linear),
{Size = UDim2.fromScale(1, 0)}
)

tween:Play()

local startTime = os.clock()

task.spawn(function()
while cooldowns[key] do
local remaining = duration - (os.clock() - startTime)

if remaining <= 0 then
break
end

refs.countdown.Text = tostring(math.ceil(remaining))
task.wait(0.1)
end
end)

if hasDied then
return
end

cooldowns[key] = false
refs.overlay.Visible = false
refs.countdown.Visible = false
end)
end

task.spawn(function()
	local layer = 4
	local fadeTime = 0
	local speed = 5
	local loops = 10

	local animation1 = Instance.new("Animation")
	animation1.AnimationId = "http://www.roblox.com/asset/?version=1&id=204062532"

	local animation2 = Instance.new("Animation")
	animation2.AnimationId = "http://www.roblox.com/asset/?version=1&id=126753849"

	local track1 = humanoid:LoadAnimation(animation1)
	local track2 = humanoid:LoadAnimation(animation2)

	SetLayer(track1, layer)
	SetLayer(track2, layer)

	track1.Looped = true
	track2.Looped = true

	track1:Play(fadeTime)
	track2:Play(fadeTime)

	track1.TimePosition = 0
	track2.TimePosition = 0

	track1:AdjustSpeed(speed)
	track2:AdjustSpeed(speed)

	local completedLoops = 0
	local lastTime = track1.TimePosition

	while completedLoops < loops do
		local currentTime = track1.TimePosition

		if currentTime < lastTime then
			completedLoops += 1
		end

		lastTime = currentTime
		task.wait()
	end

	track1:Stop(fadeTime)
	track2:Stop(fadeTime)

	animation1:Destroy()
	animation2:Destroy()
end)

task.spawn(function()
	local BOX_COUNT = 15
	local SPAWN_INTERVAL = 0.09
	local TRAVEL_DISTANCE = 20
	local TRAVEL_TIME = 0.3

	for i = 1, BOX_COUNT do
		task.spawn(function()
			local lookDirection = torso.CFrame.LookVector
			local rightDirection = torso.CFrame.RightVector
			local upDirection = Vector3.new(0, 1, 0)

			local forwardDistance = math.random(5, 15) / 10
			local randomSide = math.random(-20, 20) / 15
			local randomHeight = math.random(-20, 20) / 10

			local spawnPosition =
				torso.Position
				+ lookDirection * forwardDistance
				+ rightDirection * randomSide
				+ upDirection * randomHeight

			local box = Instance.new("Part")
			box.Name = "PurplePunch"
			box.Size = Vector3.new(1.5, 1.5, 3)
			box.CFrame = CFrame.lookAt(
				spawnPosition,
				spawnPosition + lookDirection
			)
			box.Color = Color3.fromRGB(170, 0, 255)
			box.Material = Enum.Material.Neon
			box.Anchored = true
			box.CanCollide = false
			box.CanTouch = false
			box.CanQuery = false
			box.Transparency = 0
			box.Parent = workspace

			local light = Instance.new("PointLight")
			light.Color = Color3.fromRGB(170, 0, 255)
			light.Brightness = 5
			light.Range = 10
			light.Parent = box

			local rayParams = RaycastParams.new()
			rayParams.FilterType = Enum.RaycastFilterType.Exclude
			rayParams.FilterDescendantsInstances = {character, box}

			local startPosition = spawnPosition
			local targetPosition = spawnPosition + lookDirection * TRAVEL_DISTANCE
			local hitPosition = nil

			local startTime = os.clock()

			while os.clock() - startTime < TRAVEL_TIME do
				if not box.Parent then
					return
				end

				local elapsed = math.clamp((os.clock() - startTime) / TRAVEL_TIME, 0, 1)
				local currentPosition = startPosition:Lerp(targetPosition, elapsed)
				local previousPosition = startPosition:Lerp(
					targetPosition,
					math.clamp((elapsed - 0.05), 0, 1)
				)

				local direction = currentPosition - previousPosition

				if direction.Magnitude > 0 then
					local result = workspace:Raycast(
						previousPosition,
						direction,
						rayParams
					)

					if result then
						hitPosition = result.Position
						break
					end
				end

				box.CFrame = CFrame.lookAt(
					currentPosition,
					currentPosition + lookDirection
				)

				box.Transparency = elapsed
				task.wait()
			end

			if hitPosition then
				box:Destroy()

				local sphere = Instance.new("Part")
				sphere.Name = "PurpleExplosion"
				sphere.Shape = Enum.PartType.Ball
				sphere.Size = Vector3.new(5, 5, 5)
				sphere.Position = hitPosition
				sphere.Color = Color3.fromRGB(170, 0, 255)
				sphere.Material = Enum.Material.Neon
				sphere.Anchored = true
				sphere.CanCollide = false
				sphere.CanTouch = false
				sphere.CanQuery = false
				sphere.Transparency = 0
				sphere.Parent = workspace

				local explosionLight = Instance.new("PointLight")
				explosionLight.Color = Color3.fromRGB(170, 0, 255)
				explosionLight.Brightness = 8
				explosionLight.Range = 15
				explosionLight.Parent = sphere

				local sphereTween = TweenService:Create(
					sphere,
					TweenInfo.new(
						0.25,
						Enum.EasingStyle.Quad,
						Enum.EasingDirection.Out
					),
					{
						Size = Vector3.new(7, 7, 7),
						Transparency = 1
					}
				)

				sphereTween:Play()
				sphereTween.Completed:Wait()

				sphere:Destroy()
			else
				box:Destroy()
			end
		end)

		if i < BOX_COUNT then
			task.wait(SPAWN_INTERVAL)
		end
	end
end)
end

local function playCustomAnimation(config)
    local animation = Instance.new("Animation")
    animation.AnimationId = "rbxassetid://" .. tostring(config.id)

    local track = animator:LoadAnimation(animation)
    
    local layerIndex = math.clamp(config.layer or 1, 1, #PriorityLayers)
    track.Priority = PriorityLayers[layerIndex]

    local fadeTime = config.fadeTime or 0.3
    track:Play(fadeTime)
    track.TimePosition = config.timePos or 0
    track:AdjustSpeed(config.speed or 1)

    task.delay(config.duration or 1, function()
        track:Stop(fadeTime)
    end)
end

local START_SIZE = 1
local MIN_SIZE = 2
local MAX_SIZE = 4
local SPREAD_RADIUS = 20
local MIN_SPREAD_RATIO = 0.6
local SPACING_FACTOR = 0.8
local ANGLE_JITTER = 0.15
local TILT_JITTER = 15
local ROTATION_OFFSET = 180
local EXPAND_TIME = 0.6
local START_TRANSPARENCY = 0.5
local BLOCK_COLOR = Color3.fromRGB(200, 200, 200)
local Y_OFFSET = -6
local SPAWN_INTERVAL = 0.3

local function CreateCrater(position)
local AVG_SIZE = (MIN_SIZE + MAX_SIZE) / 2
local ringRadius = SPREAD_RADIUS * ((1 + MIN_SPREAD_RATIO) / 2)
local circumference = 2 * math.pi * ringRadius
local BLOCK_COUNT = math.max(1, math.ceil(circumference / (AVG_SIZE * SPACING_FACTOR)))
for i = 1, BLOCK_COUNT do
local size = math.random(MIN_SIZE * 10, MAX_SIZE * 10) / 10
local baseAngle = (i / BLOCK_COUNT) * math.pi * 2
local angle = baseAngle + (math.random() * 2 - 1) * ANGLE_JITTER
local minDist = SPREAD_RADIUS * MIN_SPREAD_RATIO
local dist = minDist + math.random(0, (SPREAD_RADIUS - minDist) * 10) / 10
local outward = Vector3.new(math.cos(angle), 0, math.sin(angle))
local offset = outward * dist
local startPosition = position + Vector3.new(0, Y_OFFSET, 0)
local targetPosition = position + offset + Vector3.new(0, Y_OFFSET, 0)
local tiltX = math.rad(math.random(-TILT_JITTER, TILT_JITTER))
local tiltZ = math.rad(math.random(-TILT_JITTER, TILT_JITTER))
local rotation = CFrame.new(Vector3.new(), outward) * CFrame.Angles(0, math.rad(ROTATION_OFFSET), 0) * CFrame.Angles(tiltX, 0, tiltZ)
local block = Instance.new("WedgePart")
block.Size = Vector3.new(START_SIZE, START_SIZE, START_SIZE)
block.CFrame = CFrame.new(startPosition) * rotation
block.Transparency = START_TRANSPARENCY
block.Color = BLOCK_COLOR
block.Material = Enum.Material.Slate

local RunService = game:GetService("RunService")

local camera = workspace.CurrentCamera

local PriorityLayers = {
Enum.AnimationPriority.Core,
Enum.AnimationPriority.Idle,
Enum.AnimationPriority.Movement,
Enum.AnimationPriority.Action,
Enum.AnimationPriority.Action2,
Enum.AnimationPriority.Action3,
Enum.AnimationPriority.Action4
}

local function SetLayer(track, layerIndex)
layerIndex = math.clamp(layerIndex, 1, #PriorityLayers)
track.Priority = PriorityLayers[layerIndex]
end

local leftAnim = Instance.new("Animation")
leftAnim.AnimationId = "rbxassetid://93693205"
local leftTrack = animator:LoadAnimation(leftAnim)
SetLayer(leftTrack, 2)

local rightAnim = Instance.new("Animation")
rightAnim.AnimationId = "rbxassetid://88016955"
local rightTrack = animator:LoadAnimation(rightAnim)
SetLayer(rightTrack, 3)

local downAnim = Instance.new("Animation")
downAnim.AnimationId = "rbxassetid://68433924"
local downTrack = animator:LoadAnimation(downAnim)
SetLayer(downTrack, 3)

local fullyDownAnim = Instance.new("Animation")
fullyDownAnim.AnimationId = "rbxassetid://68433924"
local fullyDownTrack = animator:LoadAnimation(fullyDownAnim)
SetLayer(fullyDownTrack, 3)

local upAnim = Instance.new("Animation")
upAnim.AnimationId = "rbxassetid://165167557"
local upTrack = animator:LoadAnimation(upAnim)
SetLayer(upTrack, 2)

local currentHorizontal = "Center"
local currentVertical = "Center"

local lastDot = 0
local lastLookY = 0

local HORIZONTAL_THRESHOLD = 0.35
local UP_THRESHOLD = 0.35
local DOWN_THRESHOLD = -0.35
local FULLY_DOWN_THRESHOLD = -0.75

local BASE_FADE = 0.5
local MIN_FADE_TIME = 0.3

local renderConnection

local function StopAllAnimations()
if leftTrack.IsPlaying then leftTrack:Stop(MIN_FADE_TIME) end
if rightTrack.IsPlaying then rightTrack:Stop(MIN_FADE_TIME) end
if upTrack.IsPlaying then upTrack:Stop(MIN_FADE_TIME) end
if downTrack.IsPlaying then downTrack:Stop(MIN_FADE_TIME) end
if fullyDownTrack.IsPlaying then fullyDownTrack:Stop(MIN_FADE_TIME) end
end

renderConnection = RunService.RenderStepped:Connect(function(deltaTime)
if workspace:FindFirstChild("SCRIPT_STOP") then
StopAllAnimations()
if renderConnection then
renderConnection:Disconnect()
renderConnection = nil
end
return
end

local cameraLook = camera.CFrame.LookVector
local flatCameraLook = Vector3.new(cameraLook.X, 0, cameraLook.Z).Unit

local dot = hrp.CFrame.RightVector:Dot(flatCameraLook)
local turnSpeed = math.abs(dot - lastDot) / deltaTime
lastDot = dot

local newHorizontal = "Center"
if dot > HORIZONTAL_THRESHOLD then
newHorizontal = "Right"
elseif dot < -HORIZONTAL_THRESHOLD then
newHorizontal = "Left"
end

if newHorizontal ~= currentHorizontal then
local hSpeedFactor = math.clamp(turnSpeed / 3, 0, 1)
local fade = math.clamp(BASE_FADE * (1 - hSpeedFactor * 0.4), MIN_FADE_TIME, BASE_FADE)

if newHorizontal == "Left" then
if rightTrack.IsPlaying then rightTrack:Stop(fade) end
leftTrack:Play(fade)
leftTrack.TimePosition = 0.3
leftTrack:AdjustSpeed(0)

elseif newHorizontal == "Right" then
if leftTrack.IsPlaying then leftTrack:Stop(fade) end
rightTrack:Play(fade)
rightTrack.TimePosition = 0
rightTrack:AdjustSpeed(1)

else
if leftTrack.IsPlaying then leftTrack:Stop(fade) end
if rightTrack.IsPlaying then rightTrack:Stop(fade) end
end

currentHorizontal = newHorizontal
end

local lookY = cameraLook.Y
local pitchSpeed = math.abs(lookY - lastLookY) / deltaTime
lastLookY = lookY

local newVertical = "Center"
if lookY >= UP_THRESHOLD then
newVertical = "Up"
elseif lookY <= FULLY_DOWN_THRESHOLD then
newVertical = "FullyDown"
elseif lookY <= DOWN_THRESHOLD then
newVertical = "Down"
end

if newVertical ~= currentVertical then
local vSpeedFactor = math.clamp(pitchSpeed / 3, 0, 1)
local fade = math.clamp(BASE_FADE * (1 - vSpeedFactor * 0.4), MIN_FADE_TIME, BASE_FADE)

if newVertical == "Up" then
if downTrack.IsPlaying then downTrack:Stop(fade) end
if fullyDownTrack.IsPlaying then fullyDownTrack:Stop(fade) end
upTrack:Play(fade)
upTrack.TimePosition = 0.3
upTrack:AdjustSpeed(0)

elseif newVertical == "Down" then
if upTrack.IsPlaying then upTrack:Stop(fade) end
if fullyDownTrack.IsPlaying then fullyDownTrack:Stop(fade) end
downTrack:Play(fade)
downTrack.TimePosition = 0.75
downTrack:AdjustSpeed(0)

elseif newVertical == "FullyDown" then
if upTrack.IsPlaying then upTrack:Stop(fade) end
if downTrack.IsPlaying then downTrack:Stop(fade) end
fullyDownTrack:Play(fade)
fullyDownTrack.TimePosition = 0.65
fullyDownTrack:AdjustSpeed(0)

else
if upTrack.IsPlaying then upTrack:Stop(fade) end
if downTrack.IsPlaying then downTrack:Stop(fade) end
if fullyDownTrack.IsPlaying then fullyDownTrack:Stop(fade) end
end

currentVertical = newVertical
end
end)
