-- Build exact clone-to-original references while both hierarchies still match.
-- This must happen before clone.Parent = c, otherwise c:GetDescendants()
-- also contains the clone and identical-named accessory Handles get mispaired.
local originalForClone = {}
local function mapClonedHierarchy(originalParent, clonedParent)
    local originalChildren = originalParent:GetChildren()
    local clonedChildren = clonedParent:GetChildren()
    for index, clonedChild in ipairs(clonedChildren) do
        local originalChild = originalChildren[index]
        if originalChild then
            originalForClone[clonedChild] = originalChild
            mapClonedHierarchy(originalChild, clonedChild)
        end
    end
end

    elseif object:IsA("BasePart") then
        object.CanCollide = false
        object.CanTouch = false
        object.CanQuery = false
        object.Massless = true
        object.Anchored = true
        pcall(function() object.CollisionGroup = CLONE_GROUP end)
        table.insert(cloneParts, object)
        if object.Name == "HumanoidRootPart" then
            object.Transparency = 1
        else
            object.Transparency = 0
            local realPart = originalForClone[object]
            if realPart and realPart:IsA("BasePart") and realPart ~= hrp0 and realPart ~= hrp1 then
                table.insert(partPairs, {real = realPart, clone = object})
            end
        end
    elseif object:IsA("Decal") then
        object.Transparency = 0
    end
end

RunService.Heartbeat:Connect(function(dt)
    if not c or not c.Parent or not hrp0 or not hrp1 or not clone or not clone.Parent then return end
    
    for _, part in ipairs(cloneParts) do
        part.CanCollide = false
        part.CanTouch = false
        part.CanQuery = false
    end

    -- In shift-lock, explicitly keep the driver root aligned to the camera yaw.
    -- The visual rig is rebuilt from this root later in the same update.
    if UserInputService.MouseBehavior == Enum.MouseBehavior.LockCenter then
        local currentCamera = workspace.CurrentCamera
        if currentCamera then
            local look = currentCamera.CFrame.LookVector
            local flatLook = Vector3.new(look.X, 0, look.Z)
            if flatLook.Magnitude > 0.001 then
                local rootPosition = hrp1.Position
                hrp1.CFrame = CFrame.lookAt(rootPosition, rootPosition + flatLook.Unit)
            end
        end
    end

    local moveVel = hrp1.CFrame:VectorToObjectSpace(hrp1.AssemblyLinearVelocity)
    local horizontalSpeed = Vector3.new(moveVel.X, 0, moveVel.Z).Magnitude
    local currentlyMoving = horizontalSpeed >= IDLE_SPEED_THRESHOLD

    if currentlyMoving ~= isMoving then
        isMoving = currentlyMoving
        updateBaseIdleAnimations()
    end

    local targetExtraXDeg = 0
    local targetExtraYDeg = 0

    if modeActive then
        if isConsecutiveHolding then
            targetExtraXDeg = -43
            targetExtraYDeg = 0
        else
            targetExtraXDeg = 10
            targetExtraYDeg = currentYDeg
        end
    end

    local factorX = 1 - math.exp(-X_TRANSITION_SPEED * dt)
    local factorY = 1 - math.exp(-Y_TRANSITION_SPEED * dt)

    smoothExtraX = smoothExtraX + (math.rad(targetExtraXDeg) - smoothExtraX) * factorX
    smoothExtraY = smoothExtraY + (math.rad(targetExtraYDeg) - smoothExtraY) * factorY

    local allowPhysicsBounce = not isConsecutiveHolding

    local isGrounded = realHumanoid and (
        realHumanoid:GetState() == Enum.HumanoidStateType.Running or 
        realHumanoid:GetState() == Enum.HumanoidStateType.RunningNoPhysics or 
        realHumanoid:GetState() == Enum.HumanoidStateType.Landed
    )
    local isIdle = horizontalSpeed < IDLE_SPEED_THRESHOLD

    local targetAlpha = (isIdle and allowPhysicsBounce) and 1 or 0
    bounceAlpha = bounceAlpha + (targetAlpha - bounceAlpha) * math.clamp(BLEND_SPEED_BOUNCE * dt, 0, 1)

    local targetAir = (isGrounded and allowPhysicsBounce) and 0 or 1
    airAlpha = airAlpha + (targetAir - airAlpha) * math.clamp(BLEND_SPEED_AIR * dt, 0, 1)

    if allowPhysicsBounce then
        idleTime = idleTime + dt * IDLE_SPEED
        if isGrounded and not isIdle then
            walkTime = walkTime + dt * STEP_SPEED
        end
    end

    local idleWeight = bounceAlpha * (1 - airAlpha)
    local idlePhase = math.sin(idleTime)
    local idleY = 0
    local idlePitchX = 0

    if idlePhase > 0 then
        idleY = idlePhase * IDLE_UP * idleWeight
        idlePitchX = idlePhase * IDLE_PITCH_FORWARD * idleWeight
    else
        idleY = idlePhase * IDLE_DOWN * idleWeight
        idlePitchX = idlePhase * IDLE_PITCH_BACK * idleWeight
    end

    local walkWeight = allowPhysicsBounce and ((1 - bounceAlpha) * (1 - airAlpha)) or 0
    
    local swayPhase = math.sin(walkTime)
    local walkX = swayPhase * STEP_SWAY_WIDTH * walkWeight
    local walkY = -math.abs(swayPhase) * STEP_BOUNCE_HEIGHT * walkWeight
    local walkRollZ = math.cos(walkTime) * STEP_ROLL_TILT * walkWeight

    local pitchPhase = math.sin(walkTime)
    local walkPitchX = 0
    if pitchPhase > 0 then
        walkPitchX = pitchPhase * STEP_PITCH_FORWARD * walkWeight
    else
        walkPitchX = pitchPhase * STEP_PITCH_BACK * walkWeight
    end

    local consecSwayX = 0
    local consecRollZ = 0

    if isConsecutiveHolding then
        consecSwayTime = consecSwayTime + dt * CONSEC_SWAY_SPEED
        consecSwayX = math.sin(consecSwayTime) * CONSEC_SWAY_WIDTH
        consecRollZ = math.cos(consecSwayTime) * CONSEC_SWAY_ROLL
    else
        consecSwayTime = 0
    end

    local targetAngleX = 0
    local targetAngleZ = 0

    if allowPhysicsBounce then
        targetAngleX = math.clamp((moveVel.Z * MULT_FORWARD_VEL) + (moveVel.Y * MULT_VERTICAL_VEL), -MAX_TILT, MAX_TILT)
        targetAngleZ = math.clamp(-moveVel.X * MULT_SIDE_VEL, -MAX_TILT, MAX_TILT)

        local characterLook = hrp1.CFrame.LookVector
        local characterRight = hrp1.CFrame.RightVector
        local upVector = Vector3.new(0, 1, 0)

        local gravityPitch = characterLook:Dot(upVector)
        local gravityRoll = characterRight:Dot(upVector)

        targetAngleX = math.clamp(targetAngleX - (gravityPitch * MULT_GRAVITY_INFLUENCE), -MAX_TILT, MAX_TILT)
        targetAngleZ = math.clamp(targetAngleZ + (gravityRoll * MULT_GRAVITY_INFLUENCE), -MAX_TILT, MAX_TILT)
    end

    local forceX = (targetAngleX - currentAngleX) * TILT_STIFFNESS
    velocityX = (velocityX + forceX * dt) * math.clamp(1 - (TILT_DAMPING * dt), 0, 1)
    currentAngleX = currentAngleX + velocityX * dt

    local forceZ = (targetAngleZ - currentAngleZ) * TILT_STIFFNESS
    velocityZ = (velocityZ + forceZ * dt) * math.clamp(1 - (TILT_DAMPING * dt), 0, 1)
    currentAngleZ = currentAngleZ + velocityZ * dt

    local totalAngleX = currentAngleX + idlePitchX + walkPitchX + smoothExtraX
    local totalAngleZ = currentAngleZ + walkRollZ + consecRollZ

    local offsetPos = Vector3.new(walkX + consecSwayX, yOffset + idleY + walkY, 0)
    local baseCFrame = hrp1.CFrame * CFrame.new(offsetPos)
    local pivotCFrame = baseCFrame * CFrame.new(PIVOT_OFFSET)
    local tiltRotation = CFrame.Angles(totalAngleX, smoothExtraY, totalAngleZ)
    local calculatedCFrame = pivotCFrame * tiltRotation * CFrame.new(-PIVOT_OFFSET)

    hrp0.CFrame = calculatedCFrame
    hrp0.AssemblyLinearVelocity = hrp1.AssemblyLinearVelocity
    cloneRoot.CFrame = calculatedCFrame

    local realRootCFrame = hrp1.CFrame
    for _, pair in ipairs(partPairs) do
        local offsetFromRealRoot = realRootCFrame:ToObjectSpace(pair.real.CFrame)
        pair.clone.CFrame = calculatedCFrame * offsetFromRealRoot
    end

    if capePart and capePart.Parent and cloneTorso then
        local torsoCFrame = cloneTorso.CFrame
        local velocity = hrp1.AssemblyLinearVelocity
        local localVel = torsoCFrame:VectorToObjectSpace(velocity)
        local localGravity = torsoCFrame:VectorToObjectSpace(WORLD_DOWN)

        local gravityAngle = math.deg(math.atan2(localGravity.Z, -localGravity.Y))
        local forwardSpeed = -localVel.Z 
        local speedLift = math.clamp(forwardSpeed * 3.5, -20, 80)
        local fallSpeed = math.min(localVel.Y, 0)
        local fallLift = math.clamp(-fallSpeed * 2.5, 0, 50)
        
        local consecLift = isConsecutiveHolding and CONSEC_CAPE_LIFT or 0

        local targetCapeAngle = math.clamp(gravityAngle + speedLift + fallLift + consecLift, 0, 85)
        local lerpSpeed = math.clamp(dt * 8, 0, 1)
        currentCapeAngle = currentCapeAngle + (targetCapeAngle - currentCapeAngle) * lerpSpeed

        local attachmentCFrame = torsoCFrame * CFrame.new(CAPE_PIVOT_OFFSET)
        local rotatedCFrame = attachmentCFrame * CFrame.Angles(math.rad(-currentCapeAngle), 0, 0)
        capePart.CFrame = rotatedCFrame * CFrame.new(CAPE_TOP_OFFSET)
    end
end)
